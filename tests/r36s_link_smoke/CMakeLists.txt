cmake_minimum_required(VERSION 3.24)
project(ki_r36s_link_smoke LANGUAGES CXX)
add_library(runtime STATIC runtime.cpp)
add_library(recompiled STATIC generated.cpp)
target_link_libraries(recompiled PUBLIC runtime)
if(NOT APPLE)
  target_link_libraries(recompiled INTERFACE "$<TARGET_FILE:recompiled>")
endif()
add_library(boot STATIC boot.cpp)
target_link_libraries(boot PUBLIC runtime)
add_executable(link_smoke main.cpp)
target_link_libraries(link_smoke PRIVATE boot recompiled)
target_link_libraries(link_smoke PRIVATE "$<LINK_GROUP:RESCAN,recompiled,runtime>")
enable_testing()
add_test(NAME r36s_link_smoke COMMAND link_smoke)
